// where there are multiple asynchronous steps and the Python side is going
// to poll for completion of each step.

function _uploadFilesContinue(outputId) {
  const outputElement = document.getElementById(outputId);
  const steps = outputElement.steps;

  const next = steps.next(outputElement.lastPromiseValue);

    // step of the generator.
    outputElement.lastPromiseValue = value;
    return next.value.response;

def classificar_arquivo(nome):
    for cat, palavras_chave in categorias.items():
        if any(palavra.lower() in nome.lower() for palavra in palavras_chave):
            return cat
    return "8_Outros"

# 4. Função de Limpeza Agressiva (Reduz tokens inúteis)
def limpar_texto(texto):
    texto = re.sub(r'(?<!\n)\n(?!\n)', ' ', texto) # Remove quebras de linha no meio da frase
    texto = re.sub(r'\s+', ' ', texto) # Remove espaços duplos
    texto = re.sub(r'Diário Oficial da Cidade de São Paulo.*?Página \d+ de \d+', '', texto, flags=re.IGNORECASE)
    texto = re.sub(r'Este texto não substitui o original publicado no Diário Oficial.*?', '', texto, flags=re.IGNORECASE)
    texto = re.sub(r'\.{4,}', ' ', texto) # Remove pontilhados longos
    return texto.strip()

text_splitter = RecursiveCharacterTextSplitter.from_tiktoken_encoder(
    model_name="gpt-4",
    chunk_size=TAMANHO_CHUNK,
    chunk_overlap=OVERLAP,
    separators=["\n\n", "\n", ".", " ", ""]
)

# 5. Processamento
textos_por_categoria = {cat: [] for cat in categorias.keys()}
os.makedirs("arquivos_agrupados", exist_ok=True)

print("\nIniciando o processamento, limpeza e agrupamento...")

for nome_arquivo in uploaded.keys():
    # Pula os arquivos ignorados
    if any(ignorado.lower() in nome_arquivo.lower() for ignorado in ARQUIVOS_IGNORAR):
        print(f"-> IGNORADO (Tabela Gigante): {nome_arquivo}")
        continue
        
    if nome_arquivo.endswith(".pdf"):
        print(f"Processando: {nome_arquivo}...")
        categoria_atual = classificar_arquivo(nome_arquivo)
        
        texto_completo = ""
        with fitz.open(nome_arquivo) as pdf:
            for pagina in pdf:
                texto_completo += pagina.get_text("text") + "\n"
        
        texto_limpo = limpar_texto(texto_completo)
        chunks = text_splitter.split_text(texto_limpo)
        
        for i, chunk in enumerate(chunks):
            cabecalho = f"\n\n--- DOCUMENTO: {nome_arquivo} | PARTE: {i+1} ---\n"
            textos_por_categoria[categoria_atual].append(cabecalho + chunk)

# 6. Salvar em arquivos TXT separados e Zipar
zip_filename = "Base_IPTU_Agrupada.zip"
with zipfile.ZipFile(zip_filename, 'w') as zipf:
    for categoria, chunks in textos_por_categoria.items():
        if chunks: # Só cria o arquivo se tiver texto dentro
            nome_txt = f"arquivos_agrupados/{categoria}.txt"
            with open(nome_txt, 'w', encoding='utf-8') as f:
                f.writelines(chunks)
            zipf.write(nome_txt, arcname=f"{categoria}.txt")

print(f"\nSucesso! Arquivos separados por categoria e compactados.")
